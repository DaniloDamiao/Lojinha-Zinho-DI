// URL DO SEU GOOGLE SCRIPT (Verifique se é a versão /exec mais recente) const URL_GOOGLE = "https://script.google.com/macros/s/AKfycbxFg7XAq_moL2kFmedY7w1n3iQ2mvL3jqKZekSS_efOEGVXDAelyJZekDbxqkUiwIsE/exec"; async function sincronizarDados() { const statusDiv = document.getElementById('statusCloud'); if(statusDiv) { statusDiv.style.display = "block"; statusDiv.innerText = "☁️ Buscando dados novos..."; } try { const response = await fetch(URL_GOOGLE + "?t=" + Date.now()); const dados = await response.json(); if (dados && Array.isArray(dados)) { // Filtramos apenas itens que tenham nome para evitar o erro de 'toLowerCase' const dbFormatado = dados.filter(item => item.nome || item.Nome).map(p => { // Pegamos os valores tratando maiúsculas e minúsculas const v_custo = parseFloat(p.custo || p.Custo || 0); const v_qtd = parseInt(p.Qtd || p.qtd || p.quantidade || 0); const v_lucro = parseFloat(p.lucroacumulado || p.lucroAcumulado || 0); const v_vendidos = parseInt(p.vendidos || p.Vendidos || 0); return { nome: p.nome || p.Nome || "Produto sem nome", custo: v_custo, qtd: v_qtd, lucroAcumulado: v_lucro, vendidos: v_vendidos }; }); localStorage.setItem('estoque_db', JSON.stringify(dbFormatado)); renderizarResumo(dbFormatado); if(statusDiv) statusDiv.innerText = "✅ Sincronizado!"; } } catch (e) { console.error("Erro na sync:", e); if(statusDiv) statusDiv.innerText = "❌ Erro ao conectar"; } setTimeout(() => { if(statusDiv) statusDiv.style.display = "none"; }, 3000); } function renderizarResumo(db) { let lucroTotal = 0; let custoEstoque = 0; let custoVendido = 0; let totalVendidos = 0; db.forEach(p => { lucroTotal += (p.lucroAcumulado || 0); totalVendidos += (p.vendidos || 0); // Custo de Reposição: Custo unitário x Qtd que sobrou no estoque custoEstoque += (p.custo * p.qtd); // Custo Recuperado: Custo unitário x Qtd que já saiu custoVendido += (p.custo * p.vendidos); }); // Atualiza os elementos na tela com segurança const elLucro = document.getElementById('lucroTotal'); const elCusto = document.getElementById('custoTotal'); const elCaixa = document.getElementById('caixaTotal'); const elVendas = document.getElementById('qtdVendas'); if(elLucro) elLucro.innerText = `R$ ${lucroTotal.toFixed(2)}`; if(elCusto) elCusto.innerText = `R$ ${custoEstoque.toFixed(2)}`; if(elCaixa) elCaixa.innerText = `R$ ${(lucroTotal + custoVendido).toFixed(2)}`; if(elVendas) elVendas.innerText = `${totalVendidos} unidades vendidas`; } // Inicialização segura window.onload = function() { const dadosLocais = localStorage.getItem('estoque_db'); if (dadosLocais) { try { const db = JSON.parse(dadosLocais); renderizarResumo(db); } catch(e) { console.error("Erro no cache:", e); } } sincronizarDados(); };
